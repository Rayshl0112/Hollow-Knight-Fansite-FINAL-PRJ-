        document.addEventListener('DOMContentLoaded', function() {
            const checkboxes = document.querySelectorAll('input[name="charms"]');
            const notchCount = document.getElementById('notch-count');
            const warningElement = document.getElementById('charm-warning');
            const submitBtn = document.getElementById('submit-btn');
            const form = document.getElementById('build-form');
            
            function updateNotchCount() {
                let total = 0;
                checkboxes.forEach(checkbox => {
                    if (checkbox.checked) {
                        total += parseInt(checkbox.getAttribute('data-notches'));
                    }
                });
                notchCount.textContent = total;
                
                if (total > 11) {
                    notchCount.style.color = '#ff6b6b';
                    warningElement.style.display = 'block';
                    submitBtn.disabled = true;
                    submitBtn.style.opacity = '0.5';
                } else {
                    notchCount.style.color = '#4a6fb5';
                    warningElement.style.display = 'none';
                    submitBtn.disabled = false;
                    submitBtn.style.opacity = '1';
                }
            }
            
            checkboxes.forEach(checkbox => {
                checkbox.addEventListener('change', updateNotchCount);
            });

            const nailLevelSelect = document.getElementById('nail-level');
            const nailDamageStat = document.querySelector('.stat-item:nth-child(2) .stat-value');
            
            const nailDamageMap = {
                'old-nail': '5',
                'sharpened': '9',
                'channeled': '13',
                'coiled': '17',
                'pure': '21'
            };

            nailLevelSelect.addEventListener('change', function() {
                nailDamageStat.textContent = nailDamageMap[this.value];
            });

            form.addEventListener('submit', function(e) {
                e.preventDefault();
                
                const notchTotal = parseInt(notchCount.textContent);
                if (notchTotal > 11) {
                    alert('Cannot submit build: Charm notches exceed maximum capacity (11)!');
                    return;
                }
                
                if (notchTotal === 0) {
                    if (!confirm('No charms selected. Submit build anyway?')) {
                        return;
                    }
                }
                
                alert('Build submitted successfully! (This is a demo)');
            });
			
            updateNotchCount();
        });
    </script>
